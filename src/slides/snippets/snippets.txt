=== s25-fix jsx
function Menu() {
  const [list, setList] = useState({ items: [], page: 1 });

  const loadMore = async () => {
    const page = list.page + 1;
    const res = await api.products({ Page: page, Take: 12 });
    setList((l) => ({ items: [...l.items, ...res.products], page }));
  };

  return (
    <div className="grid">
      {list.items.map((p) => <ProductCard key={p.id} p={p} />)}
      <button onClick={loadMore}>Load more</button>
    </div>
  );
}
